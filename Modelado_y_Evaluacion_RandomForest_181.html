from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import RandomizedSearchCV, StratifiedKFold

cv = StratifiedKFold(
    n_splits=3,
    shuffle=True,
    random_state=42
)

param_dist = {
    "n_estimators": [200, 300, 500],
    "max_depth": [8, 12, 16, 20, 25],
    "min_samples_split": [5, 10, 20],
    "min_samples_leaf": [4, 8, 12, 20],
    "max_features": ["sqrt", "log2"],
    "class_weight": ["balanced", "balanced_subsample"]
}

random_search_rf_181 = RandomizedSearchCV(
    estimator=RandomForestClassifier(
        random_state=42,
        n_jobs=-1
    ),
    param_distributions=param_dist,
    n_iter=15,
    scoring="roc_auc",
    cv=cv,
    random_state=42,
    n_jobs=-1,
    verbose=1
)

random_search_rf_181.fit(X_train, y_train)

modelo_181 = random_search_rf_181.best_estimator_

print("\nMejores parámetros:")
print(random_search_rf_181.best_params_)

print(
    f"\nMejor ROC-AUC promedio CV: "
    f"{random_search_rf_181.best_score_:.4f}"
)

Decidimos ajustar el espacio de búsqueda de hiperparámetros, aumentando min_samples_split y min_samples_leaf y limitando max_depth. Además, se incrementó n_iter de 10 a 15 para explorar un mayor número de configuraciones. La estrategia de validación y la métrica de optimización (ROC-AUC) se mantuvieron constantes para permitir una comparación directa entre ambos modelos.
